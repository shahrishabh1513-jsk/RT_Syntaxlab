🔧 1. Introduction to C Getting Started

C is a powerful, general-purpose programming language created in 1972. It's the foundation of many modern languages and operating systems like Linux and Windows.

#include <stdio.h>

int main() {
    printf("Hello, World!");
    return 0;
}
Output:
Hello, World!

📊 2. Variables & Data Types Values & Types

Common data types: int, float, double, char. Variables must be declared before use.

int age = 25;
float price = 19.99f;
char grade = 'A';
double pi = 3.14159;

printf("Age: %d, Grade: %c\n", age, grade);
Output:
Age: 25, Grade: A

🧮 3. Operators Expressions

Arithmetic, relational, logical, and bitwise operators perform operations on data.

int a = 10, b = 3;
printf("%d %d %d\n", a + b, a % b, a > b);

⚖️ 4. Control Flow if / else / switch

Use if-else and switch statements for conditional execution.

int score = 65;

if (score >= 60) {
    printf("Pass");
} else {
    printf("Fail");
}

switch (3) {
    case 1: printf("One"); break;
    case 3: printf("Three"); break;
    default: printf("Other");
}
Output:
PassThree

🔄 5. Loops for / while / do-while

Repeat code blocks with for, while, and do-while loops.

for (int i = 0; i < 5; i++) {
    printf("%d ", i);
}

int i = 0;
while (i < 3) {
    printf("%d ", i);
    i++;
}

⚙️ 6. Functions Reusable Code

Functions organize code into reusable blocks that can return values and take parameters.

int add(int a, int b) {
    return a + b;
}

int main() {
    int result = add(10, 20);
    printf("Sum: %d", result);
    return 0;
}
Output:
Sum: 30

📚 7. Arrays & Strings Fixed Collections

Arrays store multiple elements of the same type. Strings in C are just character arrays.

int numbers[5] = {1, 2, 3, 4, 5};
char name[] = "SyntaxLab";

for (int i = 0; i < 5; i++) {
    printf("%d ", numbers[i]);
}
printf("\n%s", name);

📍 8. Pointers Memory Addresses

Pointers store memory addresses and enable dynamic memory management.

int x = 10;
int *ptr = &x;

printf("Value: %d\n", *ptr);   // Dereferencing
printf("Address: %p\n", (void*)ptr);

*ptr = 20;
printf("New value: %d\n", x);
Output:
Value: 10 Address: 0x... New value: 20

🏗️ 9. Structures Custom Data Types

struct groups related variables of different types under one name.

struct Student {
    char name[50];
    int age;
};

int main() {
    struct Student s1 = {"John", 20};
    printf("%s is %d\n", s1.name, s1.age);
    return 0;
}
Output:
John is 20

🧠 10. Dynamic Memory Allocation malloc / free

Allocate memory at runtime with malloc() and release it with free() to avoid memory leaks.

#include <stdlib.h>

int *arr = (int*) malloc(5 * sizeof(int));
if (arr != NULL) {
    for (int i = 0; i < 5; i++) arr[i] = i * 2;
    printf("%d", arr[2]);
    free(arr);
}
Output:
4

📄 11. File Handling Reading & Writing

Use FILE* pointers with fopen, fprintf and fscanf to work with files.

FILE *fp = fopen("data.txt", "w");
fprintf(fp, "Hello, C!");
fclose(fp);

char buffer[50];
fp = fopen("data.txt", "r");
fgets(buffer, 50, fp);
printf("%s", buffer);
fclose(fp);

#️⃣ 12. Preprocessor Directives #define / #include

Directives like #define and #include are processed before compilation.

#define PI 3.14159
#define SQUARE(x) ((x) * (x))

int main() {
    printf("%f\n", PI);
    printf("%d\n", SQUARE(5));
    return 0;
}
Output:
3.141590 25

🔗 13. Unions & Enums Alternate Data Types

union shares memory between members; enum defines named integer constants.

enum Day { MON, TUE, WED };
union Data { int i; float f; };

int main() {
    enum Day today = TUE;
    union Data d;
    d.i = 10;
    printf("%d %d", today, d.i);
    return 0;
}
Output:
1 10
